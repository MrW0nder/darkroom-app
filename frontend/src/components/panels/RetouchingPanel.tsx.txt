
import React, { useState } from 'react';
import { Brush, Copy, Droplet, Eye, Sparkles } from 'lucide-react';

type RetouchTool = 'healing' | 'clone' | 'spot' | 'redeye' | 'skin';

export const RetouchingPanel: React.FC = () => {
  const [activeTool, setActiveTool] = useState<RetouchTool>('healing');
  const [brushSize, setBrushSize] = useState(20);
  const [opacity, setOpacity] = useState(100);
  const [hardness, setHardness] = useState(80);
  const [skinSmoothStrength, setSkinSmoothStrength] = useState(50);

  const tools: { id: RetouchTool; name: string; icon: React.ReactNode; description: string }[] = [
    {
      id: 'healing',
      name: 'Healing Brush',
      icon: <Brush className="w-5 h-5" />,
      description: 'Content-aware healing for seamless corrections'
    },
    {
      id: 'clone',
      name: 'Clone Stamp',
      icon: <Copy className="w-5 h-5" />,
      description: 'Clone pixels from source to destination'
    },
    {
      id: 'spot',
      name: 'Spot Removal',
      icon: <Droplet className="w-5 h-5" />,
      description: 'Remove spots and blemishes'
    },
    {
      id: 'redeye',
      name: 'Red-Eye Correction',
      icon: <Eye className="w-5 h-5" />,
      description: 'Fix red-eye effect'
    },
    {
      id: 'skin',
      name: 'Skin Smoothing',
      icon: <Sparkles className="w-5 h-5" />,
      description: 'Professional skin retouching'
    }
  ];

  return (
    <div className="retouching-panel bg-gray-900 text-white p-4 h-full overflow-y-auto">
      <h2 className="text-xl font-bold mb-4">Retouching Tools</h2>

      {/* Tool Selection */}
      <div className="space-y-2 mb-6">
        {tools.map(tool => (
          <button
            key={tool.id}
            onClick={() => setActiveTool(tool.id)}
            className={`w-full p-3 rounded flex items-center gap-3 transition-colors ${
              activeTool === tool.id
                ? 'bg-blue-600'
                : 'bg-gray-800 hover:bg-gray-700'
            }`}
          >
            {tool.icon}
            <div className="text-left flex-1">
              <div className="font-medium">{tool.name}</div>
              <div className="text-xs text-gray-400">{tool.description}</div>
            </div>
          </button>
        ))}
      </div>

      {/* Tool Settings */}
      <div className="space-y-4">
        <h3 className="font-semibold text-sm text-gray-400">TOOL SETTINGS</h3>

        {/* Common brush settings */}
        {(activeTool === 'healing' || activeTool === 'clone' || activeTool === 'spot') && (
          <>
            <div>
              <label className="block text-sm mb-2">
                Brush Size: {brushSize}px
              </label>
              <input
                type="range"
                min="1"
                max="200"
                value={brushSize}
                onChange={(e) => setBrushSize(parseInt(e.target.value))}
                className="w-full"
              />
            </div>

            <div>
              <label className="block text-sm mb-2">
                Opacity: {opacity}%
              </label>
              <input
                type="range"
                min="0"
                max="100"
                value={opacity}
                onChange={(e) => setOpacity(parseInt(e.target.value))}
                className="w-full"
              />
            </div>

            <div>
              <label className="block text-sm mb-2">
                Hardness: {hardness}%
              </label>
              <input
                type="range"
                min="0"
                max="100"
                value={hardness}
                onChange={(e) => setHardness(parseInt(e.target.value))}
                className="w-full"
              />
            </div>
          </>
        )}

        {/* Clone Stamp specific */}
        {activeTool === 'clone' && (
          <div className="bg-gray-800 p-3 rounded">
            <p className="text-sm text-gray-400">
              Hold Alt/Option and click to set source point
            </p>
          </div>
        )}

        {/* Red-Eye specific */}
        {activeTool === 'redeye' && (
          <div>
            <label className="block text-sm mb-2">
              Correction Intensity: {opacity}%
            </label>
            <input
              type="range"
              min="0"
              max="100"
              value={opacity}
              onChange={(e) => setOpacity(parseInt(e.target.value))}
              className="w-full"
            />
            <p className="text-xs text-gray-400 mt-2">
              Click and drag over red eyes to correct
            </p>
          </div>
        )}

        {/* Skin Smoothing specific */}
        {activeTool === 'skin' && (
          <>
            <div>
              <label className="block text-sm mb-2">
                Smoothing Strength: {skinSmoothStrength}%
              </label>
              <input
                type="range"
                min="0"
                max="100"
                value={skinSmoothStrength}
                onChange={(e) => setSkinSmoothStrength(parseInt(e.target.value))}
                className="w-full"
              />
            </div>

            <div className="flex items-center gap-2">
              <input
                type="checkbox"
                id="preserve-texture"
                defaultChecked
                className="w-4 h-4"
              />
              <label htmlFor="preserve-texture" className="text-sm">
                Preserve Texture
              </label>
            </div>

            <div>
              <label className="block text-sm mb-2">Blur Radius: 5px</label>
              <input
                type="range"
                min="1"
                max="20"
                defaultValue="5"
                className="w-full"
              />
            </div>

            <button className="w-full py-2 bg-blue-600 rounded mt-2">
              Apply to Selection
            </button>
          </>
        )}
      </div>

      {/* Quick Tips */}
      <div className="mt-6 bg-gray-800 p-3 rounded">
        <h4 className="font-semibold text-sm mb-2">Quick Tips</h4>
        <ul className="text-xs text-gray-400 space-y-1">
          <li>• Use healing brush for seamless texture blending</li>
          <li>• Clone stamp for precise pixel duplication</li>
          <li>• Adjust opacity for subtle corrections</li>
          <li>• Lower hardness for softer edges</li>
        </ul>
      </div>
    </div>
  );
};
